+def _js_tokens(text):
+    """Lex JS conservatively; ambiguous syntax yields no source exemptions.
+
+    Comments, strings, regexes and whole templates remain opaque. Interpolations
+    are traversed only to locate the end of a template, never to grant exemptions.
+    """
+    tokens = []
+    size = len(text)

+    def scan(pos, interpolation=False):
+        depth, previous = 0, ""
+        while pos < size:
+            start, char = pos, text[pos]
+            if char.isspace():
+                pos += 1
+                continue
+            if text.startswith("//", pos):
+                end = text.find("\n", pos)
+                pos = size if end < 0 else end
+                continue
+            if text.startswith("/*", pos):
+                end = text.find("*/", pos + 2)
+                if end < 0:
+                    raise ValueError("unterminated comment")
+                pos = end + 2
+                continue
+            kind = "punct"
+            if char in "'\"`":
+                kind = "template" if char == "`" else "string"
+                pos += 1
+                while pos < size:
+                    if text[pos] == "\\":
+                        pos += 2
+                    elif text[pos] == char:
+                        pos += 1
+                        break
+                    elif char == "`" and text.startswith("${", pos):
+                        pos = scan(pos + 2, True)
+                    elif char != "`" and text[pos] in "\r\n":
+                        raise ValueError("unterminated string")
+                    else:
+                        pos += 1
+                else:
+                    raise ValueError("unterminated literal")
+            elif char == "/":
+                # These require statement/expression grammar to disambiguate.
+                if previous in {"", ")", "}"}:
+                    raise ValueError("ambiguous slash")
+                operand = bool(re.fullmatch(r"[A-Za-z_$][A-Za-z0-9_$]*|[0-9]+", previous))

+                    "instanceof", "in", "of", "new", "await", "else", "do",
+                }
+                if (operand and not keyword) or previous in {"literal", "]"}:
+                    pos += 1
+                else:
+                    kind, bracket = "regex", False
+                    pos += 1
+                    while pos < size and text[pos] not in "\r\n":

+                        if text[pos] == "[":

+                        elif text[pos] == "]":
+                            bracket = False
+                        elif text[pos] == "/" and not bracket:
+                            pos += 1
+                            while pos < size and text[pos].isalpha():

+def source_is_safe(text, path, known_secrets=()):
+    """Validate exact source bytes separately from general log/prompt redaction.
+
+    Only complete JS references, storage-key lookups and runtime Bearer templates
+    can override labelled-value matches. Other literal detectors and inherited
+    credentials remain enforced. This does not prove runtime data flow safety.
+    """
+    for secret in known_secrets:
+        if secret and any(value in text for value in (
+            secret, json.dumps(secret)[1:-1], json.dumps(secret, ensure_ascii=False)[1:-1]
+        )):
+            return False
+    if redact_secret_text(text) == text:
+        return True
+    if Path(path).suffix.lower() not in {".js", ".mjs", ".cjs"}:
+        return False
+    tokens = _js_tokens(text)
+    starts = {item[0]: index for index, item in enumerate(tokens)}
+    ends = {item[1] for item in tokens}
+    masked = list(text)
+    for match in _SECRET_TEXT_RE.finditer(text):
+        index = starts.get(match.start())
+        if index is None or index + 2 >= len(tokens):
+            continue
+        key, operator = tokens[index], tokens[index + 1]
+        if key[2] not in {"word", "string"} or key[3].strip("'\"") != match.group(2):
+            continue
+        previous = tokens[index - 1][3] if index else ""
+        if not (
+            (operator[3] == ":" and previous in {"{", ","})
+            or (operator[3] == "=" and previous in {"const", "let", "var", ";", "{"})
+        ):
+            continue
+        rhs_start = tokens[index + 2][0]
+        expression = _JS_SAFE_RHS.match(text, rhs_start)
+        if expression is None:
+            continue

+            rhs_end -= 1
+        if rhs_end not in ends:
+            continue
+        masked[match.start():match.end()] = " " * (match.end() - match.start())
+    view = "".join(masked)
+    return redact_secret_text(view) == view
